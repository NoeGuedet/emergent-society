# C1.1 — Journal Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Build the canonical per-node journal — the single truth of the system — with hash-chained append-only events, framed zstd persistence, claim-check blobs, bounded write-behind with an explicit flush barrier, reader-side chain verification and torn-tail repair.

**Architecture:** One readable canonical log per node (`nodes/<uid>/journal.v0.jsonl.zstd`), everything else is a disposable index. Events are enveloped (`v`, `type`, `seq`, `time`, `prev_hash`, `hash`, `ignorable`, `data`), canonicalized with RFC 8785 (JCS) and chained with SHA-256. Durable batches are written as independent length-prefixed zstd frames (one `write` + `fsync` per batch); the hot path never blocks on I/O. Repair is the reader's job; only a physically torn trailing fragment is discarded.

**Tech Stack:** TypeScript (strict) on Node 24 LTS, Vitest, `canonicalize` (RFC 8785), `node:crypto` (SHA-256), `node:zlib` zstd (built-in since Node 22.15), `node:fs`.

**Spec:** [`../kernel.md`](../kernel.md) §3 (the journal), §1.6 (the journal is below the physical/behavioral frontier), §8 (`unhandledRejection` posture is out of scope here — it belongs to C1.7).

## Global Constraints

- Node **24 LTS**, TypeScript **strict mode** from the first line. `"type": "module"`, ESM only.
- Dependencies are pinned to **exact versions** (no `^`, no `~`). Runtime deps allowed in C1.1: `canonicalize` only. Dev deps: `typescript`, `vitest`, `@types/node`.
- The event envelope format is **frozen before the first write**: `v: 0` is reserved from the genesis event. Any later format change means `v: 1` and a new file, never an in-place mutation.
- **Append-only.** No code path may rewrite or truncate journaled events; the only permitted destructive operation is discarding a physically torn trailing fragment during repair.
- The discriminated union of event types is **extensible by declaration merging** (interface `EventDataMap {}`) — `assertNever` is forbidden; a reader that meets an unknown non-ignorable type **refuses to rebuild**.
- All timestamps go through an injectable `now(): number` — never a direct `Date.now()` call inside the journal modules (clock virtualization from the start).
- Commit after every task. Commit messages in English, conventional style (`feat:`, `test:`, `chore:`).

```ts
// src/journal/canon.ts
import { createHash } from 'node:crypto';
import canonicalize from 'canonicalize';

export type JsonValue =
  | null
  | boolean
  | number
  | string
  | JsonValue[]
  | { [k: string]: JsonValue };

export class NonCanonicalizableError extends Error {
  constructor(reason: string) {
    super(`value is not canonicalizable: ${reason}`);
    this.name = 'NonCanonicalizableError';
  }
}

export function canonicalizeJson(value: JsonValue): string {
  const result = canonicalize(value, (key, val) => {
    if (val === undefined) throw new NonCanonicalizableError(`undefined at key "${key}"`);
    if (typeof val === 'number' && !Number.isFinite(val)) {
      throw new NonCanonicalizableError(`non-finite number at key "${key}"`);
    }
    return val;
  });
  if (result === undefined) throw new NonCanonicalizableError('top-level undefined');
  if (typeof value === 'number' && !Number.isFinite(value)) {
    throw new NonCanonicalizableError('non-finite top-level number');
  }
  return result;
}

export function sha256Hex(input: string): string {
  return createHash('sha256').update(input, 'utf8').digest('hex');
}
```

```ts
// src/journal/envelope.ts
import { canonicalizeJson, sha256Hex, type JsonValue } from './canon.js';

export const FORMAT_VERSION = 0;
export const GENESIS_HASH = '0'.repeat(64);

/** Extended by declaration merging in later plans. */
export interface EventDataMap {}
export type EventType = keyof EventDataMap | (string & {});

export interface EventEnvelope<T extends JsonValue = JsonValue> {
  v: number;
  type: string;
  seq: number;
  time: number;
  prev_hash: string;
  hash: string;
  ignorable?: boolean;
  data: T;
}

export function computeHash(e: Omit<EventEnvelope, 'hash'>): string {
  const payload: Record<string, JsonValue> = {
    v: e.v, type: e.type, seq: e.seq, time: e.time, data: e.data,
  };
  if (e.ignorable !== undefined) payload['ignorable'] = e.ignorable;
  return sha256Hex(e.prev_hash + canonicalizeJson(payload as JsonValue));
}

export function makeEvent(input: {
  type: string; data: JsonValue; seq: number; time: number; prevHash: string; ignorable?: boolean;
}): EventEnvelope {
  const unsigned: Omit<EventEnvelope, 'hash'> = {
    v: FORMAT_VERSION, type: input.type, seq: input.seq, time: input.time,
    prev_hash: input.prevHash, data: input.data,
    ...(input.ignorable !== undefined ? { ignorable: input.ignorable } : {}),
  };
  return { ...unsigned, hash: computeHash(unsigned) };
}

export function verifyEvent(e: EventEnvelope): boolean {
  const { hash, ...unsigned } = e;
  return computeHash(unsigned) === hash;
}

export class UnknownEventTypeError extends Error {
  constructor(public readonly eventType: string) {
    super(`unknown non-ignorable event type: ${eventType} — refusing to rebuild`);
    this.name = 'UnknownEventTypeError';
  }
}

export function assertKnownType(
  type: string, ignorable: boolean, known: ReadonlySet<string>,
): void {
  if (!known.has(type) && !ignorable) throw new UnknownEventTypeError(type);
}
```

```ts
// src/journal/blobs.ts
import { createHash } from 'node:crypto';
import { mkdir, open, readFile, writeFile } from 'node:fs/promises';
import { join } from 'node:path';

/** Format constant — a payload at or beyond this size is stored as a blob. */
export const CLAIM_CHECK_THRESHOLD = 16 * 1024;

export class BlobStore {
  constructor(private readonly home: string) {}

  private pathFor(hash: string): string {
    return join(this.home, 'blobs', hash.slice(0, 2), hash);
  }

  async put(content: Buffer): Promise<string> {
    const hash = createHash('sha256').update(content).digest('hex');
    const path = this.pathFor(hash);
    await mkdir(join(this.home, 'blobs', hash.slice(0, 2)), { recursive: true });
    // wx: never overwrite an existing blob — content addressing makes it immutable.
    const handle = await open(path, 'wx').catch(() => null);
    if (handle) {
      await handle.write(content);
      await handle.sync();
      await handle.close();
    }
    return hash;
  }

  async get(hash: string): Promise<Buffer> {
    return readFile(this.pathFor(hash));
  }

  async has(hash: string): Promise<boolean> {
    try {
      await readFile(this.pathFor(hash));
      return true;
    } catch {
      return false;
    }
  }
}
```

### Task 5: Framed zstd codec (persistence format)

One durable batch = `[u32 LE compressed length][one independent zstd frame]`; the frame decompresses to the batch's canonical JSONL lines. The length prefix gives header-only listing and exact torn-tail detection. (Deviation from "pure concatenated frames", documented: `node:zlib` cannot enumerate frame boundaries without an external size table.)

**Files:**
- Create: `src/journal/framing.ts`
- Test: `src/journal/__tests__/framing.test.ts`

**Interfaces:**
- Produces:
  - `encodeBatch(lines: string[]): Buffer` — joins with `\n`, zstd-compresses (`zstdCompressSync`), prepends the 4-byte LE length.
  - `decodeBatch(frame: Buffer): string[]`
  - `scanBatches(buf: Buffer): { batches: Buffer[]; tornBytes: number }` — walks length-prefixed frames; a trailing fragment (length announces more bytes than remain, or an undecodable frame) is reported as `tornBytes`, never thrown.

```ts
// src/journal/framing.ts
import { zstdCompressSync, zstdDecompressSync } from 'node:zlib';

const LEN_BYTES = 4;

export function encodeBatch(lines: string[]): Buffer {
  const compressed = zstdCompressSync(Buffer.from(lines.join('\n'), 'utf8'));
  const out = Buffer.alloc(LEN_BYTES + compressed.length);
  out.writeUInt32LE(compressed.length, 0);
  compressed.copy(out, LEN_BYTES);
  return out;
}

export function decodeBatch(frame: Buffer): string[] {
  const compressed = frame.subarray(LEN_BYTES, LEN_BYTES + frame.readUInt32LE(0));
  return zstdDecompressSync(compressed).toString('utf8').split('\n');
}

export function scanBatches(buf: Buffer): { batches: Buffer[]; tornBytes: number } {
  const batches: Buffer[] = [];
  let offset = 0;
  while (offset + LEN_BYTES <= buf.length) {
    const len = buf.readUInt32LE(offset);
    const end = offset + LEN_BYTES + len;
    if (end > buf.length) return { batches, tornBytes: buf.length - offset };
    const frame = buf.subarray(offset, end);
    try {
      zstdDecompressSync(frame.subarray(LEN_BYTES));
    } catch {
      return { batches, tornBytes: buf.length - offset };
    }
    batches.push(frame);
    offset = end;
  }
  return { batches, tornBytes: buf.length - offset };
}
```

### Task 6: The writer (append, batching, flush barrier, single ownership)

**Files:**
- Create: `src/journal/writer.ts`
- Test: `src/journal/__tests__/writer.test.ts`
- Consumes: `makeEvent`, `verifyEvent`, `EventEnvelope`, `GENESIS_HASH` (Task 3); `BlobStore`, `CLAIM_CHECK_THRESHOLD` (Task 4); `encodeBatch` (Task 5).

**Interfaces:**
- Consumes: `scanBatches` from `src/journal/framing.ts` (Task 5) for repair-on-open (Task 7 wires it fully; Task 6 uses it to resume `seq`/`prev_hash`).
- Produces:
  - `class SessionAlreadyOwnedError extends Error`
  - `class JournalWriter` with:
    - `static async open(home: string, nodeUid: string, opts?: { now?: () => number; batchWindowMs?: number }): Promise<JournalWriter>`
    - `append(type: string, data: JsonValue, opts?: { ignorable?: boolean }): EventEnvelope` — synchronous, in-memory; enqueues into the current burst.
    - `flush(): Promise<void>` — **the barrier**: writes the pending burst as one batch (one `write` + `fsync`), updates `journal.v0.head`. Called before every model request and every top-level tool effect (callers land in C1.3/C1.4).
    - `close(): Promise<void>` — flush + release the lock.
  - Lock: `nodes/<uid>/journal.v0.lock` created `wx` with the pid inside; if it exists and the pid is dead (`process.kill(pid, 0)` throws `ESRCH`), the lock is taken over; otherwise `SessionAlreadyOwnedError`.
  - Claim-check in `append`: if `canonicalizeJson(data).length >= CLAIM_CHECK_THRESHOLD`, store the serialized payload in the `BlobStore` and replace `data` with `{ blob: <hash>, size: <bytes> }`; payloads that must be truncated are marked `{ truncated: true, size }` — never silently.
  - Write-behind: the first `append` of a burst starts a `batchWindowMs` (default 200) timer; on fire it writes the batch. The hot path never blocks on I/O.

```ts
// src/journal/writer.ts
import { appendFile, mkdir, open as openFile, readFile, rm, writeFile } from 'node:fs/promises';
import { join } from 'node:path';
import { canonicalizeJson, type JsonValue } from './canon.js';
import { GENESIS_HASH, makeEvent, verifyEvent, type EventEnvelope } from './envelope.js';
import { BlobStore, CLAIM_CHECK_THRESHOLD } from './blobs.js';
import { encodeBatch, scanBatches, decodeBatch } from './framing.js';

export class SessionAlreadyOwnedError extends Error {
  constructor(nodeUid: string) {
    super(`journal for node "${nodeUid}" is already owned by a live writer`);
    this.name = 'SessionAlreadyOwnedError';
  }
}

interface Head { first_hash: string; last_hash: string; count: number; ts: number }

export class JournalWriter {
  private seq = 0;
  private prevHash = GENESIS_HASH;
  private firstHash: string | null = null;
  private pending: EventEnvelope[] = [];
  private pendingBlobs: Promise<string>[] = [];
  private timer: NodeJS.Timeout | null = null;
  private closed = false;

  private constructor(
    private readonly dir: string,
    private readonly blobs: BlobStore,
    private readonly now: () => number,
    private readonly batchWindowMs: number,
  ) {}

  static async open(
    home: string, nodeUid: string,
    opts: { now?: () => number; batchWindowMs?: number } = {},
  ): Promise<JournalWriter> {
    const dir = join(home, 'nodes', nodeUid);
    await mkdir(dir, { recursive: true });
    await acquireLock(dir, nodeUid);
    const w = new JournalWriter(
      dir, new BlobStore(home),
      opts.now ?? (() => Date.now()), opts.batchWindowMs ?? 200,
    );
    await w.resume();
    return w;
  }

  private get logPath() { return join(this.dir, 'journal.v0.jsonl.zstd'); }
  private get headPath() { return join(this.dir, 'journal.v0.head'); }
  private get lockPath() { return join(this.dir, 'journal.v0.lock'); }

  private async resume(): Promise<void> {
    let raw: Buffer;
    try { raw = await readFile(this.logPath); } catch { return; }
    const { batches } = scanBatches(raw);
    for (const frame of batches) {
      for (const line of decodeBatch(frame)) {
        const e = JSON.parse(line) as EventEnvelope;
        this.seq = e.seq + 1;
        this.prevHash = e.hash;
        this.firstHash ??= e.hash;
      }
    }
  }

  append(type: string, data: JsonValue, opts: { ignorable?: boolean } = {}): EventEnvelope {
    if (this.closed) throw new Error('writer is closed');
    const stored = this.claimCheck(data);
    const e = makeEvent({
      type, data: stored, seq: this.seq, time: this.now(), prevHash: this.prevHash,
      ...(opts.ignorable !== undefined ? { ignorable: opts.ignorable } : {}),
    });
    this.seq += 1;
    this.prevHash = e.hash;
    this.firstHash ??= e.hash;
    this.pending.push(e);
    this.timer ??= setTimeout(() => { void this.flush(); }, this.batchWindowMs);
    return e;
  }

  private claimCheck(data: JsonValue): JsonValue {
    const canonical = canonicalizeJson(data);
    if (canonical.length < CLAIM_CHECK_THRESHOLD) return data;
    const hash = createHashSync(canonical);
    this.pendingBlobs.push(this.blobs.put(Buffer.from(canonical, 'utf8')));
    return { blob: hash, size: canonical.length };
  }

  async flush(): Promise<void> {
    if (this.timer) { clearTimeout(this.timer); this.timer = null; }
    // Referenced blobs must be durable before the events that reference them.
    await Promise.all(this.pendingBlobs.splice(0));
    if (this.pending.length === 0) return;
    const lines = this.pending.map((e) => canonicalizeJson(e as unknown as JsonValue));
    this.pending = [];
    await appendFile(this.logPath, encodeBatch(lines));
    const head: Head = {
      first_hash: this.firstHash ?? GENESIS_HASH,
      last_hash: this.prevHash,
      count: this.seq,
      ts: this.now(),
    };
    await writeFile(this.headPath, JSON.stringify(head));
    const handle = await openFile(this.logPath, 'r');
    await handle.sync();
    await handle.close();
  }

  async close(): Promise<void> {
    if (this.closed) return;
    await this.flush();
    this.closed = true;
    await rm(this.lockPath, { force: true });
  }
}

function createHashSync(s: string): string {
  return createHash('sha256').update(s, 'utf8').digest('hex');
}

async function acquireLock(dir: string, nodeUid: string): Promise<void> {
  const lockPath = join(dir, 'journal.v0.lock');
  try {
    await writeFile(lockPath, String(process.pid), { flag: 'wx' });
  } catch {
    const pid = Number(await readFile(lockPath, 'utf8'));
    try {
      process.kill(pid, 0);
      throw new SessionAlreadyOwnedError(nodeUid);
    } catch (err) {
      if (err instanceof SessionAlreadyOwnedError) throw err;
      await rm(lockPath, { force: true }); // stale lock: previous owner is dead
      await writeFile(lockPath, String(process.pid), { flag: 'wx' });
    }
  }
}
```

```ts
// src/journal/reader.ts
import { open as openFile, readFile, truncate } from 'node:fs/promises';
import { join } from 'node:path';
import { GENESIS_HASH, assertKnownType, verifyEvent, type EventEnvelope } from './envelope.js';
import { scanBatches, decodeBatch } from './framing.js';

export class ChainBreakError extends Error {
  constructor(public readonly seq: number, reason: string) {
    super(`hash chain broken at seq ${seq}: ${reason}`);
    this.name = 'ChainBreakError';
  }
}

export interface Head { first_hash: string; last_hash: string; count: number; ts: number }

export class JournalReader {
  private constructor(
    private readonly dir: string,
    private readonly known: ReadonlySet<string>,
  ) {}

  static async open(
    home: string, nodeUid: string, knownTypes: ReadonlySet<string>,
  ): Promise<JournalReader> {
    return new JournalReader(join(home, 'nodes', nodeUid), knownTypes);
  }

  private get logPath() { return join(this.dir, 'journal.v0.jsonl.zstd'); }

  async head(): Promise<Head | null> {
    try {
      return JSON.parse(await readFile(join(this.dir, 'journal.v0.head'), 'utf8')) as Head;
    } catch {
      return null;
    }
  }

  async *events(fromSeq = 0): AsyncGenerator<EventEnvelope> {
    let raw: Buffer;
    try { raw = await readFile(this.logPath); } catch { return; }
    const { batches } = scanBatches(raw);
    let prevHash = GENESIS_HASH;
    let expectedSeq = 0;
    for (const frame of batches) {
      for (const line of decodeBatch(frame)) {
        const e = JSON.parse(line) as EventEnvelope;
        if (e.seq !== expectedSeq) throw new ChainBreakError(e.seq, `expected seq ${expectedSeq}`);
        if (e.prev_hash !== prevHash) throw new ChainBreakError(e.seq, 'prev_hash mismatch');
        if (!verifyEvent(e)) throw new ChainBreakError(e.seq, 'hash recomputation failed');
        assertKnownType(e.type, e.ignorable ?? false, this.known);
        prevHash = e.hash;
        expectedSeq += 1;
        if (e.seq >= fromSeq) yield e;
      }
    }
  }
}

export async function repair(home: string, nodeUid: string): Promise<{ tornBytes: number }> {
  const path = join(home, 'nodes', nodeUid, 'journal.v0.jsonl.zstd');
  let raw: Buffer;
  try { raw = await readFile(path); } catch { return { tornBytes: 0 }; }
  const { tornBytes } = scanBatches(raw);
  if (tornBytes > 0) {
    await truncate(path, raw.length - tornBytes);
    const handle = await openFile(path, 'r');
    await handle.sync();
    await handle.close();
  }
  return { tornBytes };
}
```
